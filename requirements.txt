# Core runtime -- everything needed to serve the API and process rasters.
#
# rasterio ships GDAL inside its wheels, so no system GDAL package is needed and
# the container stays on python:3.12-slim. Development and analysis extras live
# in requirements-dev.txt so they are not baked into the image.

# processing/ -- pure geospatial computation, no web dependencies (PLAN.md 10)
rasterio>=1.5.0
numpy>=2.0

# catalogue/ -- AOI and footprint intersection
shapely>=2.0

# cog.validate_cog; falls back to a structural check when absent
rio-cogeo>=5.0

# backend/ -- API layer
fastapi>=0.115
uvicorn[standard]>=0.30

# backend/db/ -- scene metadata cache (PLAN.md 6). Required to import the API
# even when BHOOMI_DATABASE_URL is unset; Postgres itself stays optional, the
# library does not. psycopg[binary] ships its own libpq, so the image needs no
# system postgres client package.
SQLAlchemy>=2.0
alembic>=1.13
psycopg[binary]>=3.2

# backend/queue/ -- the job queue (PLAN.md 4.2). RQ rather than Celery: one
# broker, no result backend to configure, and the worker is a process rather
# than a framework.
rq>=2.0
redis>=5.0

# backend/storage.py -- object storage for outputs (PLAN.md D14: Cloudflare R2).
# boto3 rather than an R2-specific client, because R2 is reached through the S3
# API; the same code runs against S3 and MinIO with a different endpoint.
boto3>=1.35
